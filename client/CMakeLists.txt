include(FetchContent)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(SDL_FORCE_STATIC_VCRT ON CACHE BOOL "" FORCE)
set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_TEST OFF CACHE BOOL "" FORCE)
set(GLAD_PROFILE "core" CACHE STRING "" FORCE)
set(GLAD_API "gl=3.3" CACHE STRING "" FORCE)
set(GLAD_REPRODUCIBLE ON CACHE BOOL "" FORCE)
FetchContent_Declare(SDL2 URL https://www.libsdl.org/release/SDL2-2.30.9.tar.gz)
FetchContent_Declare(glm URL https://github.com/g-truc/glm/archive/refs/tags/0.9.9.8.tar.gz)
FetchContent_Declare(glad URL https://github.com/Dav1dde/glad/archive/refs/tags/v0.1.36.tar.gz)
FetchContent_MakeAvailable(SDL2 glm glad)
file(GLOB CLIENT_SOURCES CONFIGURE_DEPENDS src/*.cpp)
add_executable(MineCppClone ${CLIENT_SOURCES})
target_compile_features(MineCppClone PRIVATE cxx_std_17)
target_compile_definitions(MineCppClone PRIVATE GLM_ENABLE_EXPERIMENTAL)
target_include_directories(MineCppClone PRIVATE include "${glm_SOURCE_DIR}/glm")
target_link_libraries(MineCppClone PRIVATE SDL2::SDL2main SDL2::SDL2-static glad glm)
if(MINGW)
    target_link_options(MineCppClone PRIVATE -static-libgcc -static-libstdc++)
endif()
add_custom_command(TARGET MineCppClone POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/textures" "$<TARGET_FILE_DIR:MineCppClone>/textures"
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/shaders" "$<TARGET_FILE_DIR:MineCppClone>/shaders")
install(TARGETS MineCppClone RUNTIME DESTINATION .)
install(DIRECTORY textures shaders DESTINATION .)
install(FILES README.md ORIGIN.md ORIGIN-PANORAMA.md VANILLA-REFERENCE.md BUILDING.md DESTINATION .)

if(MC_BUILD_TESTS)
    add_executable(client_movement_tests tests/movement_tests.cpp src/movement.cpp)
    target_compile_features(client_movement_tests PRIVATE cxx_std_17)
    target_include_directories(client_movement_tests PRIVATE include "${glm_SOURCE_DIR}/glm")
    target_link_libraries(client_movement_tests PRIVATE glm)
    add_test(NAME client_movement_tests COMMAND client_movement_tests)
    add_executable(client_settings_tests tests/settings_tests.cpp src/settings.cpp)
    target_compile_features(client_settings_tests PRIVATE cxx_std_17)
    target_include_directories(client_settings_tests PRIVATE include)
    add_test(NAME client_settings_tests COMMAND client_settings_tests)
endif()
